const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Content-Type": "application/json; charset=utf-8", "Cache-Control": "no-store" }; export default { async fetch(request, env, ctx) { try { if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders }); } const url = new URL(request.url); if (url.pathname !== "/fund") { return json({ success: false, message: "Not found. Please use /fund?codes=000001" }, 404); } const codesParam = url.searchParams.get("codes") || ""; const codes = codesParam .split(/[,，\s\n]+/) .map(x => x.trim()) .filter(Boolean); if (!codes.length) { return json({ success: false, message: "Missing codes parameter. Example: /fund?codes=000001,012348" }, 400); } const results = []; for (const code of codes) { const result = await fetchSingleFund(code); results.push(result); } return json({ success: true, count: results.length, data: results, results: results, fetchedAt: new Date().toISOString() }); } catch (err) { return json({ success: false, message: "Worker fatal error", error: err && err.message ? err.message : String(err) }, 500); } } }; async function fetchSingleFund(code) { try { if (!/^\d{6}$/.test(code)) { return { success: false, code, fundcode: code, message: "Invalid fund code. Fund code should be 6 digits." }; } const apiUrl = `https://fundgz.1234567.com.cn/js/${code}.js?rt=${Date.now()}`; const resp = await fetch(apiUrl, { headers: { "User-Agent": "Mozilla/5.0", "Referer": "https://fund.eastmoney.com/" } }); const text = await resp.text(); const match = text.match(/jsonpgz\((.*)\);?/); if (!match) { return { success: false, code, fundcode: code, message: "Unexpected API response", raw: text.slice(0, 300) }; } const data = JSON.parse(match[1]); return { success: true, // 标准字段，前端优先读取这些 code: data.fundcode || code, name: data.name || "", navDate: data.jzrq || "", unitNav: data.dwjz || "", estimateNav: data.gsz || "", estimateRate: data.gszzl || "", updateTime: data.gztime || "", // 保留天天基金原始字段，兼容旧前端 fundcode: data.fundcode || code, jzrq: data.jzrq || "", dwjz: data.dwjz || "", gsz: data.gsz || "", gszzl: data.gszzl || "", gztime: data.gztime || "", source: "天天基金公开实时估值接口" }; } catch (err) { return { success: false, code, fundcode: code, message: err && err.message ? err.message : String(err) }; } } function json(obj, status = 200) { return new Response(JSON.stringify(obj, null, 2), { status, headers: corsHeaders }); }
